{{- /* Citation block: a plain-text reference and BibTeX. In order of priority: 1. `bibtex` in front matter, used verbatim; 2. `arxiv` in front matter, cited as the arXiv entry; 3. otherwise the blog post itself. */ -}} {{- $names := slice }} {{- range .Params.authors }}{{ $names = $names | append .name }}{{ end }} {{- with $names }} {{- /* BibTeX joins authors with " and "; the prose line reads as a sentence. */}} {{- $authors := delimit . " and " }} {{- $prose := cond (eq (len .) 2) (delimit . " and ") (delimit . ", " ", and ") }} {{- $parts := split (index . 0) " " }} {{- $last := index $parts (sub (len $parts) 1) | lower }} {{- $word := index (split ($.Title | lower | replaceRE `[^a-z0-9 ]` "") " ") 0 }} {{- $arxiv := partial "arxiv-id.html" $ }} {{- $year := $.Date.Format "2006" }} {{- /* New-style arXiv IDs start with YYMM. */}} {{- if findRE `^\d{4}\.\d{4,5}$` $arxiv }}{{ $year = printf "20%s" (substr $arxiv 0 2) }}{{ end }} {{- $key := $.Params.citekey | default (printf "%s%s%s" $last $year $word) }} {{- $bib := "" }} {{- if $.Params.bibtex }} {{- $bib = $.Params.bibtex }} {{- else if $arxiv }} {{- $lines := slice (printf " title = {%s}," $.Title) (printf " author = {%s}," $authors) (printf " year = {%s}," $year) (printf " eprint = {%s}," $arxiv) " archivePrefix = {arXiv}," }} {{- with $.Params.arxiv_class }}{{ $lines = $lines | append (printf " primaryClass = {%s}," .) }}{{ end }} {{- $lines = $lines | append (printf " url = {https://arxiv.org/abs/%s}" $arxiv) }} {{- $bib = printf "@misc{%s,\n%s\n}" $key (delimit $lines "\n") }} {{- else }} {{- $bib = printf "@misc{%s,\n title = {%s},\n author = {%s},\n year = {%s},\n month = %s,\n howpublished = {\\url{%s}},\n note = {%s blog}\n}" $key $.Title $authors $year (lower ($.Date.Format "Jan")) $.Permalink site.Title }} {{- end }}

{{ $prose }}. “{{ $.Title }}.” arXiv:{{ $arxiv }}, {{ $year }}. https://arxiv.org/abs/{{ $arxiv }}

{{- else }}

{{ $prose }}. “{{ $.Title }}.” {{ site.Title }} blog, {{ $.Date.Format "January 2006" }}. {{ $.Permalink }}
